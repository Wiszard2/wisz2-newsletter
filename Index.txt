  // ---------------- signup logic ----------------
  var PENDING_KEY = 'wisz2_pending_signup_v1';

  function escapeHtml(s){
    return String(s).replace(/[&<>"']/g, function(c){
      return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c];
    });
  }

  function readState(){
    try {
      var el = document.getElementById('wisz2-state');
      if (!el) return { signups: [] };
      var parsed = JSON.parse(el.textContent);
      if (!parsed || !Array.isArray(parsed.signups)) return { signups: [] };
      return parsed;
    } catch (e) {
      return { signups: [] };
    }
  }

  var STATE = readState();

  function setTally(){
    var el = document.getElementById('tally-count');
    if (el) el.textContent = String(STATE.signups.length);
  }
  setTally();

  // Builds the FULL replacement document (doctype-first) for artifact.publish().
  // This must stay structurally identical to the authored page, parameterized by state.
  function renderFullDocument(state){
    var headHtml = document.head.innerHTML;
    var bodyHtml = document.body.innerHTML.replace(
      /<script id="wisz2-state"[^>]*>[\s\S]*?<\/script>/,
      ''
    );
    var stateScript = '<script id="wisz2-state" type="application/json">' +
      JSON.stringify(state).replace(/</g, '\\u003c') +
      '</' + 'script>';
    return '<!doctype html><html><head>' + headHtml + '</head><body>' +
      bodyHtml + stateScript + '</body></html>';
  }

  function showStatus(kind, msg){
    var el = document.getElementById('form-status');
    el.className = 'form-status show ' + kind;
    el.textContent = msg;
  }

  function setBusy(busy){
    var btn = document.getElementById('submit-btn');
    btn.disabled = busy;
    btn.textContent = busy ? 'Joining…' : 'Get the Weekly Briefing';
  }

  async function attemptPublish(entry, artifactApi){
    var newState = { signups: STATE.signups.concat([entry]) };
    try {
      await artifactApi.publish(renderFullDocument(newState));
      // success: this view reloads automatically per the artifact contract
    } catch (err) {
      if (err && err.code === 'conflict') {
        // shell is already reloading every view to the latest version;
        // the pending entry stays in sessionStorage and we retry after reload.
        return;
      }
      throw err;
    }
  }

  async function retryPendingOnLoad(){
    var raw;
    try { raw = sessionStorage.getItem(PENDING_KEY); } catch (e) { raw = null; }
    if (!raw) return;
    var pending;
    try { pending = JSON.parse(raw); } catch (e) { try { sessionStorage.removeItem(PENDING_KEY); } catch(e2){} return; }

    var alreadyIn = STATE.signups.some(function(s){ return s.email === pending.email && s.ts === pending.ts; });
    if (alreadyIn){
      try { sessionStorage.removeItem(PENDING_KEY); } catch(e){}
      showStatus('ok', "You're on the list — Issue 01 lands Sep 8.");
      return;
    }

    var artifactApi;
    try { artifactApi = await claude.use('artifact'); } catch (e) { artifactApi = null; }
    if (!artifactApi) return;

    try {
      showStatus('wait', 'Finishing sign-up…');
      await attemptPublish(pending, artifactApi);
    } catch (e) {
      // leave pending in sessionStorage; user can resubmit manually if needed
    }
  }

  document.getElementById('signup-form').addEventListener('submit', async function(ev){
    ev.preventDefault();
    var name = document.getElementById('f-name').value.trim();
    var email = document.getElementById('f-email').value.trim();
    var company = document.getElementById('f-company').value.trim();
    var role = document.getElementById('f-role').value;

    if (!name || !email){
      showStatus('err', 'Please add your name and work email.');
      return;
    }

    var entry = { name: name, email: email, company: company, role: role, ts: Date.now() };

    var artifactApi;
    try { artifactApi = typeof claude !== 'undefined' ? await claude.use('artifact') : null; } catch (e) { artifactApi = null; }

    if (!artifactApi){
      showStatus('err', 'Sign-up is temporarily unavailable here — email matt.wisz@wisz2.com and we\'ll add you directly.');
      return;
    }

    try { sessionStorage.setItem(PENDING_KEY, JSON.stringify(entry)); } catch (e) {}

    setBusy(true);
    showStatus('wait', 'Adding you to the list…');

    try {
      await attemptPublish(entry, artifactApi);
      // on success the view reloads; nothing else to do here
    } catch (err) {
      setBusy(false);
      var code = err && err.code;
      if (code === 'not_writer' || code === 'not_granted' || code === 'consent_required'){
        showStatus('err', 'This view is read-only. Email matt.wisz@wisz2.com to be added directly.');
      } else if (code === 'rate_limited'){
        showStatus('err', 'A lot of sign-ups just came through — please try again in a few seconds.');
      } else {
        showStatus('err', 'Something went wrong. Please try again, or email matt.wisz@wisz2.com.');
      }
    }
  });

  document.getElementById('final-cta').addEventListener('click', function(){
    document.getElementById('f-name').focus();
    document.getElementById('f-name').scrollIntoView({ behavior: 'smooth', block: 'center' });
  });

  retryPendingOnLoad();
})();
